module http  -- HTTP client, request records and strict query decoding.  (use str)
  notes: Client: verified HTTPS, plaintext HTTP only on loopback. ASCII URLs with valid escapes; no userinfo or fragments. Methods GET HEAD POST PUT PATCH DELETE OPTIONS. Reserved headers: Host, Content-Length, Transfer-Encoding, Connection. No redirects, proxies or network examples. Limits: 5 s connect, 8 MiB request headers+body and response body, 64 KiB/100 response headers. Response body is strict UTF-8; headers are sorted lowercase names, retaining duplicates. Server setup: lm help serve.
*type http.Request  (rec (method Str) (target Str) (query Str) (headers (List (Pair Str Str))) (body Str))
*type http.Response  (rec (status Int) (headers (List (Pair Str Str))) (body Str))
*error http.BadEncoding  (s Str)
*error http.HttpError  (url Str) (msg Str)
*fn http.fetch  (url Str) -> Response  [eff net.http]  [fails HttpError]  -- GET with empty body, no caller headers and a 30-second total deadline.
 fn http.query-fields  (fields (List Str)) -> (List (Pair Str Str))  [fails BadEncoding]  -- Decode nonempty query fields in order.
*fn http.query-pairs  (q Str) -> (List (Pair Str Str))  [fails BadEncoding]  -- Decode query fields, keeping duplicates and replacing plus with space.
*fn http.request  (method Str) (url Str) (headers (List (Pair Str Str))) (body Str) -> Response  [eff net.http]  [fails HttpError]  -- Send one request with a 30-second total deadline. Non-2xx statuses are results.
*fn http.request-within  (method Str) (url Str) (headers (List (Pair Str Str))) (body Str) (secs Int) -> Response  [eff net.http]  [fails HttpError]  -- Send one request; a deadline outside 1..600 seconds fails before network I/O.
*fn http.url-decode  (s Str) -> Str  [fails BadEncoding]  -- Decode percent escapes as strict UTF-8; keep plus signs.
